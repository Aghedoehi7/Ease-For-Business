generator client {
  provider = "prisma-client-js"
  output   = "../node_modules/@prisma/client/.prisma"
}

datasource db {
  provider = "sqlite"
  url      = "file:./dev.db"
}

model User {
  id            String   @id @default(cuid())
  email         String   @unique
  password      String
  name          String?
  businessName  String?
  createdAt     DateTime @default(now())
  updatedAt     DateTime @updatedAt

  products      Product[]
  transactions   Transaction[]
  sessions       Session[]
  chatMessages   ChatMessage[]
}

model Product {
  id          String   @id @default(cuid())
  name        String
  description String?
  price       Float
  quantity    Int
  sku         String   @unique
  category    String?
  userId      String
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt

  user        User     @relation(fields: [userId], references: [id])
}

model Transaction {
  id            String   @id @default(cuid())
  userId        String
  type          String   // 'in', 'out', 'adjustment'
  reason        String?
  items         String   // JSON string of items
  totalAmount   Float
  paymentMethod String?
  createdAt     DateTime @default(now())

  user User @relation(fields: [userId], references: [id])
}

model Expense {
  id        String   @id @default(cuid())
  amount    Float
  category  String
  note      String?
  createdAt DateTime @default(now())
}

model ChatMessage {
  id        String   @id @default(cuid())
  userId    String
  role      String
  content   String
  createdAt DateTime @default(now())

  user      User     @relation(fields: [userId], references: [id])
}

model Session {
  id        String   @id @default(cuid())
  token     String   @unique
  userId    String
  expiresAt DateTime

  user      User     @relation(fields: [userId], references: [id])
}
